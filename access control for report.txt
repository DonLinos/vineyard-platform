// Shared access control for report pages.
// Two ways in: (1) Apo's real Supabase login (agronomist), or
// (2) a per-person access code (winery clients), entered on access.html
// and kept in sessionStorage for the rest of the browser tab's life.
//
// Usage in a page:
//   const access = await requireAccess(supa, WINERY_ID);
//   if (!access) return; // requireAccess already redirected
//   // access.personName -> string to show/record as "done by"
//   // access.mode -> 'auth' | 'code'

async function requireAccess(supa, wineryId) {
  const { data: { session } } = await supa.auth.getSession();
  if (session && session.user) {
    return { mode: 'auth', personName: session.user.email };
  }

  const raw = sessionStorage.getItem('accessSession');
  if (raw) {
    try {
      const s = JSON.parse(raw);
      if (s && s.wineryId === wineryId && s.personName) {
        return { mode: 'code', personName: s.personName };
      }
    } catch (e) { /* fall through to redirect */ }
  }

  const here = window.location.pathname.split('/').pop();
  window.location.href = 'access.html?winery_id=' + encodeURIComponent(wineryId || '') + '&return=' + encodeURIComponent(here);
  return null;
}
